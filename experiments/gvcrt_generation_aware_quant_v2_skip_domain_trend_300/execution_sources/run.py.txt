"""Serial preflight/evaluation on GPU7, three fixed concurrent training workers."""
import argparse, csv, fcntl, hashlib, json, os, shutil, subprocess, sys, time
from pathlib import Path
P=Path(__file__).resolve().parent
V1=P.parent/'gvcrt_generation_aware_quant_visual_condition_v1_trend_300'
PYTHON='/Huang_group/zyyz/home_dir/.conda/envs/gvc-rt/bin/python'
BRANCHES=['A','H00','H10','H01','H11']

def sha(path):
    h=hashlib.sha256()
    with open(path,'rb') as f:
        for b in iter(lambda:f.read(1<<20),b''):h.update(b)
    return h.hexdigest()

def free(gpu):
    lines=subprocess.check_output(['nvidia-smi','--query-gpu=index,uuid','--format=csv,noheader'],text=True).splitlines()
    uuid=next(x.split(',')[1].strip() for x in lines if int(x.split(',')[0])==gpu)
    apps=subprocess.check_output(['nvidia-smi','--query-compute-apps=gpu_uuid,pid','--format=csv,noheader'],text=True)
    return uuid not in apps

def run(command,gpu,log):
    lock=open(P/'server_artifacts'/f'gpu{gpu}.lock','w')
    fcntl.flock(lock,fcntl.LOCK_EX)
    while not free(gpu):
        print('QUEUE',gpu,flush=True);time.sleep(20)
    with log.open('a') as out:
        result=subprocess.run([PYTHON,*command],stdout=out,stderr=subprocess.STDOUT)
    fcntl.flock(lock,fcntl.LOCK_UN)
    if result.returncode:raise RuntimeError((command,result.returncode,str(log)))

def reuse():
    provenance=[]
    for b,old in [('A','A'),('H00','B')]:
        for src in sorted((V1/'server_artifacts/evaluation').glob(old+'_*.json')):
            r=json.loads(src.read_text())
            if not r.get('all_64_reference_states_synchronized'):continue
            ckpt=V1/'server_artifacts'/old/f"step_{r['step']}.pt"
            assert sha(ckpt)==r['checkpoint_sha256'] and sha(r['stream_path'])==r['stream_sha256']
            assert r['frames']==64 and r['independent_decode'] and r['IP']==-1
            key=f"{b}_s{r['step']}_{r['sequence']}_q{r['external_qp']}"
            dest=P/'server_artifacts/evaluation'/f'{key}.json'
            if dest.exists():continue
            frames=V1/'server_artifacts/evaluation_frames'/src.with_suffix('.csv').name
            transitions=V1/'server_artifacts/evaluation_transitions'/src.with_suffix('.csv').name
            rows=list(csv.DictReader(frames.open()));assert len(rows)==64
            r.update(branch=b,measurement='reused_v1_measurement',reuse_source=str(src),reuse_source_sha256=sha(src),
                     P_payload_bytes=sum(int(x['payload_bytes']) for x in rows[1:]),
                     actual_P_payload_bits=sum(int(x['payload_bytes']) for x in rows[1:])*8,
                     evaluation_metric_files_sha256={str(q):sha(q) for q in [P/'evaluate.py',V1/'evaluate.py',P.parents[1]/'experiments/gvcrt_neural_wrapper_v4_progressive_joint/eval_core.py']})
            r['estimated_P_total_bits']=r['estimated_P_main_latent_bits']+r['estimated_P_hyper_latent_bits']
            for kind,path in [('evaluation_frames',frames),('evaluation_transitions',transitions)]:
                (P/'server_artifacts'/kind).mkdir(exist_ok=True);shutil.copyfile(path,P/'server_artifacts'/kind/f'{key}.csv')
            dest.write_text(json.dumps(r,indent=2)+'\n')
            provenance.append({'target':str(dest),'source':str(src),'sha256':sha(src),'checkpoint_sha256':sha(ckpt),'frames_sha256':sha(frames),'transitions_sha256':sha(transitions)})
            (P/'previews').mkdir(exist_ok=True)
            for img in (V1/'previews').glob(src.stem+'*.png'):shutil.copyfile(img,P/'previews'/img.name.replace(old+'_',b+'_',1))
            gt=V1/'previews'/f"{r['sequence']}_gt_frame15.png"
            if gt.exists():shutil.copyfile(gt,P/'previews'/gt.name)
    prior=json.loads((P/'reuse_provenance.json').read_text()) if (P/'reuse_provenance.json').exists() else []
    (P/'reuse_provenance.json').write_text(json.dumps(prior+provenance,indent=2)+'\n')

def evaluate(b,step,source,qp):
    dest=P/'server_artifacts/evaluation'/f'{b}_s{step}_{source}_q{qp}.json'
    if dest.exists() and json.loads(dest.read_text()).get('status')=='PASS':return
    run([str(P/'evaluate.py'),'--branch',b,'--step',str(step),'--source',source,'--qp',str(qp),'--gpu','7'],7,P/'server_artifacts/evaluation.log')
    print('EVALUATED',b,step,source,qp,flush=True)

def supplements():
    # Select solely from measured actual-bpp endpoints, with a fixed maximum of three extra QPs.
    pairs=[('A',h) for h in BRANCHES[1:]]+[('H00','H10'),('H00','H01'),('H10','H11'),('H01','H11')]
    decisions=[]
    for source in ['UVal0','UVal1','ParkScene','Jockey']:
        for attempt in range(3):
            rows=[json.loads(f.read_text()) for f in (P/'server_artifacts/evaluation').glob(f'*_s300_{source}_q*.json')]
            points={b:[r for r in rows if r['branch']==b] for b in BRANCHES}
            targets=set()
            for a,b in pairs:
                aa,bb=points[a],points[b]
                alo,ahi=min(r['actual_bpp'] for r in aa),max(r['actual_bpp'] for r in aa)
                blo,bhi=min(r['actual_bpp'] for r in bb),max(r['actual_bpp'] for r in bb)
                if max(alo,blo)<=min(ahi,bhi):continue
                low,high=(a,b) if ahi<blo else (b,a)
                for branch,direction in [(low,'higher_rate'),(high,'lower_rate')]:
                    used={r['external_qp'] for r in points[branch]}
                    if len(used-{2,7})>=3:continue
                    candidates=([9,8,6,5,4,3,1,0] if direction=='higher_rate' else [0,1,3,4,5,6,8,9])
                    qp=next((q for q in candidates if q not in used),None)
                    if qp is not None:targets.add((branch,qp,direction))
            if not targets:break
            # Only one additional point per method in each iteration.
            chosen={}
            for b,q,d in sorted(targets):chosen.setdefault(b,(q,d))
            for b,(q,d) in chosen.items():
                evidence={x:[{'qp':r['external_qp'],'bpp':r['actual_bpp']} for r in points[x]] for x in BRANCHES}
                decisions.append({'source':source,'method':b,'qp':q,'direction':d,'rates_only_evidence':evidence})
                (P/'supplement_selection.json').write_text(json.dumps(decisions,indent=2)+'\n')
                evaluate(b,300,source,q)

def main():
    import concurrent.futures
    (P/'server_artifacts/evaluation').mkdir(parents=True,exist_ok=True)
    (P/'server_artifacts/logs').mkdir(exist_ok=True)
    run([str(P/'preflight.py')],7,P/'server_artifacts/preflight.log')
    reuse()
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
        futures=[pool.submit(run,[str(P/'train.py'),'--branch',b,'--gpu',str(g),'--stop','300'],g,P/'server_artifacts/logs'/f'{b}.log') for b,g in [('H10',4),('H01',5),('H11',6)]]
        # GPU7 works on verified baseline files while GPUs4-6 train.
        for b in ['A','H00']:
            for step in [0,100,300]:
                for source in (['UVal0','UVal1','ParkScene','Jockey'] if step==300 else ['UVal0','UVal1']):
                    for q in [2,7]:evaluate(b,step,source,q)
        for f in futures:f.result()
    for b in ['H10','H01','H11']:
        for step in [0,100,300]:
            for source in (['UVal0','UVal1','ParkScene','Jockey'] if step==300 else ['UVal0','UVal1']):
                for q in [2,7]:evaluate(b,step,source,q)
    supplements()
    subprocess.run([PYTHON,str(P/'finalize.py')],check=True)
    print('ALL_COMPLETE',flush=True)

if __name__=='__main__':main()
